```
docker run -d \
  --name quant-postgres \
  -e POSTGRES_USER=postgres \
  -e POSTGRES_PASSWORD=postgres123 \
  -e POSTGRES_DB=QuantTrading \
  -p 5432:5432 \
  -v /Users/feng/docker-data/postgres16:/var/lib/postgresql/data \
  postgres:16
```

``` json
"ConnectionStrings": {
  "Default": "Host=localhost;Port=5432;Database=QuantTrading;Username=postgres;Password=postgres123"
}
```

## 初始化 Vite + React + TypeScript 项目
```
# 1. 创建项目 (这里项目名取为 quant-trading-pwa)
npm create vite@latest quant-trading-pwa -- --template react-ts

# 2. 进入目录并安装基础依赖
cd quant-trading-pwa
npm install
```
## 安装并配置 PWA 核心插件
```
# 安装 vite-plugin-pwa 插件
npm install vite-plugin-pwa -D
```

打开项目根目录的 `vite.config.ts` 文件，进行如下配置：
```typescript
import { defineConfig } from 'vite';
import react from '@vitejs/plugin-react';
import { VitePWA } from 'vite-plugin-pwa';

export default defineConfig({
  plugins: [
    react(),
    VitePWA({
      // 内部工具建议设为自动更新，只要刷新页面就拉取新版本
      registerType: 'autoUpdate',
      
      // 包含哪些静态资源进行离线缓存
      includeAssets: ['favicon.ico', 'apple-touch-icon.png', 'safari-pinned-tab.svg'],
      
      // PWA 的清单文件配置 (这就是能被添加到手机桌面的关键)
      manifest: {
        name: 'Alpha Capital 交易复盘系统',
        short_name: '交易复盘',
        description: '内部量化交易与手动复盘看板',
        theme_color: '#ffffff', // 你的 UI 主题色
        background_color: '#f3f4f6',
        display: 'standalone', // 沉浸式全屏体验，隐藏浏览器地址栏
        icons: [
          {
            src: 'pwa-192x192.png', // 你需要准备一张 192x192 的 Logo 放在 public 目录下
            sizes: '192x192',
            type: 'image/png'
          },
          {
            src: 'pwa-512x512.png', // 你需要准备一张 512x512 的 Logo 放在 public 目录下
            sizes: '512x512',
            type: 'image/png'
          }
        ]
      }
    })
  ],
  // 解决开发环境跨域调用后端 API 的问题
  server: {
    proxy: {
      '/api': {
        target: 'http://localhost:5000', // 你的 ABP 后端地址
        changeOrigin: true,
        secure: false,
      }
    }
  }
});
```

### 将 `App.tsx` 替换为以下代码进行测试：
```typescript
import { ConfigProvider, Button, Card, DatePicker, message } from 'antd';
import { StockOutlined } from '@ant-design/icons';
// Tailwind 样式已经在 main.tsx 中通过 index.css 引入了

function App() {
  const [messageApi, contextHolder] = message.useMessage();

  const handleTradeClick = () => {
    messageApi.success('交易复盘记录已生成！');
  };

  return (
    // ConfigProvider 用于全局配置 Ant Design 的主题和语言
    <ConfigProvider
      theme={{
        token: {
          colorPrimary: '#1677ff', // 你的品牌主色调
          borderRadius: 6,         // 全局组件圆角
        },
      }}
    >
      {/* 必须保留 contextHolder 才能正常显示全局提示 */}
      {contextHolder}

      {/* 这里开始使用 Tailwind 控制宏观布局：全屏、灰底、居中 */}
      <div className="min-h-screen bg-gray-100 flex items-center justify-center p-4">
        
        {/* 使用 Ant Design 的 Card 组件作为容器 */}
        <Card 
          title="Alpha Capital 复盘面板" 
          className="w-full max-w-md shadow-lg"
          // Ant Design 的组件可以完美兼容 Tailwind 的 className
        >
          {/* 使用 Tailwind 控制组件间的垂直间距 */}
          <div className="flex flex-col gap-4">
            
            <p className="text-gray-600">
              请选择要复盘的交易日期：
            </p>
            
            {/* Ant Design 的日期选择器 */}
            <DatePicker className="w-full" />

            {/* Ant Design 的按钮配合图标 */}
            <Button 
              type="primary" 
              size="large" 
              icon={<StockOutlined />}
              onClick={handleTradeClick}
            >
              生成盈亏报表
            </Button>

          </div>
        </Card>

      </div>
    </ConfigProvider>
  );
}

export default App;
```

### 设置中文语言包 (极其重要)
在你的 `main.tsx`中加入本地化配置：
```TypeScript
// src/main.tsx
import React from 'react'
import ReactDOM from 'react-dom/client'
import App from './App.tsx'
import './index.css'
import 'virtual:pwa-register'

// 1. 引入 Ant Design 中文语言包
import zhCN from 'antd/locale/zh_CN';
import { ConfigProvider } from 'antd';
// 2. 引入 dayjs 的中文包 (Ant Design v5 依赖 dayjs 处理时间)
import 'dayjs/locale/zh-cn'; 

ReactDOM.createRoot(document.getElementById('root')!).render(
  <React.StrictMode>
    {/* 3. 在最外层包裹 ConfigProvider 并传入 locale */}
    <ConfigProvider locale={zhCN}>
      <App />
    </ConfigProvider>
  </React.StrictMode>,
)
```
## 安装和配置路由
### 安装 React Router
> npm install react-router-dom

### 配置路由字典 (修改 `main.tsx`)
``` TypeScript
import React from 'react'
import ReactDOM from 'react-dom/client'
// 1. 引入 React Router 的核心 API
import { createBrowserRouter, RouterProvider } from 'react-router-dom'

// 引入 Ant Design 配置
import zhCN from 'antd/locale/zh_CN';
import { ConfigProvider } from 'antd';
import 'dayjs/locale/zh-cn'; 

// 全局样式
import './index.css'
import 'virtual:pwa-register'

// 2. 引入你的页面组件
import App from './App.tsx'       // 作为主页/仪表盘
import Login from './Login.tsx'   // 作为登录页

// 3. 创建路由字典：定义路径和对应渲染的页面
const router = createBrowserRouter([
  {
    path: "/",
    element: <App />, // 当访问 http://localhost:5173/ 时，显示 App 面板
  },
  {
    path: "/login",
    element: <Login />, // 当访问 http://localhost:5173/login 时，显示 Login 页面
  }
]);

ReactDOM.createRoot(document.getElementById('root')!).render(
  <React.StrictMode>
    <ConfigProvider locale={zhCN}>
      {/* 4. 把原来的 <App /> 替换为路由提供者，交由它来根据 URL 决定渲染谁 */}
      <RouterProvider router={router} />
    </ConfigProvider>
  </React.StrictMode>,
)
```
